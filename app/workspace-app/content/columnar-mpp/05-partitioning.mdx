## Many files, one table

<Scene widget="pt-log">
<Step>
</Step>
<Step>

A log in a few lines of Python: each entry is a JSON file that lists data files, and a reader
reads exactly one entry's files.

```python needs=by-day
import json, os
import duckdb

LOG = "partitioning_log"
os.makedirs(LOG, exist_ok=True)
day = lambda d: f"partitioning_by_day/day=2026-08-0{d}/data_0.parquet"

def commit(version, files):
    """A new version of the table is one small file: the list of its data files."""
    with open(f"{LOG}/v{version}.json", "w") as entry:
        json.dump({"version": version, "files": files}, entry)

def rows(version):
    """A reader opens one entry, then reads exactly the files it lists."""
    with open(f"{LOG}/v{version}.json") as entry:
        files = json.load(entry)["files"]
    return duckdb.sql(f"SELECT count(*) FROM read_parquet({files})").fetchall()[0][0]

commit(0, [day(1)])
commit(1, [day(1), day(2)])
for version in (0, 1):
    print(f"v{version}: {rows(version):,} rows")
```

</Step>
