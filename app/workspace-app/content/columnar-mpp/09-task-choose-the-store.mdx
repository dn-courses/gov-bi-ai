## The tool

Run this first. It measures any statement on either store and prints the same three lines.

```python id=measure
# The measuring tool: the same three numbers for any statement, on either store.
#   measure_pg(sql)          PostgreSQL: runs it 3 times in a transaction that is rolled back
#   measure_bq(sql)          BigQuery: runs it once for real, cache off; "{sandbox}" is your own dataset
#   per_month(measured, n)   what n runs a month cost
#   run_pg(sql)              PostgreSQL: runs a statement once and keeps it (CREATE INDEX, DROP INDEX)
import json
import os
import statistics
import time
from pathlib import Path
from typing import NamedTuple

import psycopg
from google.api_core.exceptions import GoogleAPICallError
from google.cloud import bigquery

MB = 2**20  # the unit PostgreSQL prints and BigQuery bills in
PAGE_BYTES = 8192
PRICE_PER_TIB = 6.25  # BigQuery on demand, US
MAX_BYTES_BILLED = 5 * 10**9  # BigQuery refuses, free of charge, a statement that would bill more
# PostgreSQL has no price per query: a managed server is rented by the month, whatever it runs.
# Cloud SQL for PostgreSQL, Enterprise edition, Iowa (us-central1), one zone, on demand:
# $0.0413 a vCPU-hour, $0.007 a GiB-hour of memory, SSD $0.000232877 a GiB-hour ($0.17 a GiB-month).
# https://cloud.google.com/sql/pricing, read 2026-11-23.
VCPUS, MEMORY_GIB, DISK_GIB, HOURS_A_MONTH = 4, 16, 20, 730
SERVER_A_MONTH = (VCPUS * 0.0413 + MEMORY_GIB * 0.007 + DISK_GIB * 0.000232877) * HOURS_A_MONTH
LAB = "enjoytech-bq-lab.gbfs.lab_station_status"  # the BigQuery copy of station_status

KEY = json.loads(Path(os.environ["GOOGLE_APPLICATION_CREDENTIALS"]).read_text())
# student-p01@… owns sandbox_p01, student-agents-test@… owns sandbox_agents_test
SANDBOX = "sandbox_" + KEY["client_email"].split("@")[0].removeprefix("student-").replace("-", "_")
BQ = bigquery.Client.from_service_account_info(KEY)


class Measured(NamedTuple):
    store: str
    mb: float  # data read
    seconds: float  # time to answer
    dollars: float  # the price of one run; 0 on PostgreSQL, where the server is paid by the month
    note: str


def show(measured):
    seconds = measured.seconds
    took = f"{seconds:,.2f} s" if seconds >= 1 else f"{seconds * 1000:,.3f} ms"
    cost = f"${SERVER_A_MONTH:,.2f} a month for the server, however many runs"
    if measured.store == "BigQuery":
        cost = f"${measured.dollars:.6f} a run"
    print(f"{measured.store}\n  data read  {measured.mb:,.2f} MB\n  time       {took}  ({measured.note})")
    print(f"  cost       {cost}")
    return measured


def explain(cursor, sql):
    """One run: its page visits (found in memory or read from disk) and its execution time."""
    cursor.execute("EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) " + sql)
    top = cursor.fetchone()[0][0]
    return top["Plan"]["Shared Hit Blocks"] + top["Plan"]["Shared Read Blocks"], top["Execution Time"] / 1000


def measure_pg(sql, hide=(), runs=3):
    """Measure on PostgreSQL and change nothing. `hide` names indexes to measure without."""
    with psycopg.connect() as conn, conn.cursor() as cursor:
        for index in hide:
            cursor.execute(f"DROP INDEX {index}")  # undone by the rollback; the table is locked meanwhile
        took = [explain(cursor, sql) for _ in range(runs)]
        conn.rollback()
    pages = took[-1][0]
    seconds = statistics.median(seconds for _, seconds in took)
    note = f"median of {runs} runs; {pages:,} page visits of 8 kB, a page counted each time it is visited"
    return show(Measured("PostgreSQL", pages * PAGE_BYTES / MB, seconds, 0.0, note))


def measure_bq(sql):
    """Run on BigQuery for real, once. A CREATE TABLE or an UPDATE is measured the same way."""
    config = bigquery.QueryJobConfig(use_query_cache=False, maximum_bytes_billed=MAX_BYTES_BILLED)
    job = BQ.query(sql.replace("{sandbox}", SANDBOX), job_config=config)
    try:
        job.result()
    except GoogleAPICallError as refused:  # the reason in one line, without the library's traceback
        raise RuntimeError(refused.message) from None
    billed = job.total_bytes_billed or 0
    seconds = (job.ended - job.started).total_seconds()
    note = "one run; bytes billed, at least 10 MB for each table read"
    return show(Measured("BigQuery", billed / MB, seconds, billed / 2**40 * PRICE_PER_TIB, note))


def per_month(measured, runs):
    """What `runs` runs a month cost: a price per run on BigQuery, a rented server on PostgreSQL."""
    if measured.store == "BigQuery":
        print(f"{runs:,} runs a month on BigQuery: ${runs * measured.dollars:,.2f}")
        return
    busy = runs * measured.seconds
    print(f"{runs:,} runs a month on PostgreSQL: ${SERVER_A_MONTH:,.2f}, the server's rent whatever it runs;")
    print(f"  together they take {busy:,.1f} s of the month's {HOURS_A_MONTH * 3600:,} s")


def run_pg(sql):
    """Run one statement on PostgreSQL and keep its effect; print how long it took."""
    started = time.perf_counter()
    with psycopg.connect(autocommit=True) as conn:
        conn.execute(sql)
    seconds = time.perf_counter() - started
    print(f"PostgreSQL: done in {seconds:,.1f} s")
    return seconds


print(f"your BigQuery dataset: {SANDBOX}")
```

How to read it: one question on each store, and what a thousand runs of it a month would cost.

```python id=demo
COUNT_PG = "SELECT count(*) FROM station_status WHERE system_id = 'NYC'"
COUNT_BQ = f"SELECT count(*) FROM `{LAB}` WHERE system_id = 'NYC'"
per_month(measure_pg(COUNT_PG), 1_000)
per_month(measure_bq(COUNT_BQ), 1_000)
```

- **data read.** On BigQuery, the bytes billed. On PostgreSQL, every page of 8 kB the query visits,
  in memory or on disk: a page visited twice counts twice, so an index can report more than the
  whole table.
- **time.** On PostgreSQL the median of three runs; on BigQuery one run, which varies from run to run.
- **cost.** On PostgreSQL the server's rent, the same for one run or a million, and how many seconds
  of the month the runs keep it busy. On BigQuery the price of one run, and of all the runs a month.

## What you can build: the syntax

These are examples of the syntax, **not answers**: the columns are placeholders.

A B-tree index on PostgreSQL. Start every name with `ho_`; the clean-up at the end removes those.
`INCLUDE` adds columns that are stored in the index but not sorted by.

```py
# Example only: col_a, col_b, col_c are placeholders.
run_pg("CREATE INDEX ho_example ON station_status (col_a, col_b)")
run_pg("CREATE INDEX ho_example2 ON station_status (col_a, col_b) INCLUDE (col_c)")
run_pg("DROP INDEX ho_example")          # an index you no longer want
measure_pg(SQL, hide=["ho_example2"])    # measure as if that index did not exist
```

A copy on BigQuery, in your own dataset. `measure_bq` prints what building it billed and took.

```py
# Example only: col_a and the column list are placeholders.
measure_bq("""
CREATE OR REPLACE TABLE `enjoytech-bq-lab.{sandbox}.ho_example`
PARTITION BY DATE(captured_at)
CLUSTER BY col_a
AS SELECT station_id, system_id, captured_at, bikes_available
FROM `enjoytech-bq-lab.gbfs.lab_station_status`
""")
```

- A copy must **name its columns**: `SELECT *` would bill the whole 6.33 GB, and the tool refuses
  anything above 5 GB.
- Partitions come by `HOUR`, `DAY` (`DATE(captured_at)`), `MONTH` or `YEAR`
  (`TIMESTAMP_TRUNC(captured_at, MONTH)`), or by an integer range (`RANGE_BUCKET`). There are no
  weekly partitions.
- `CLUSTER BY` takes up to four columns. A summary table is a copy built with `GROUP BY`.
- Start every table name with `ho_`.

## Clean up

Remove your indexes and your tables: everything whose name starts with `ho_`.

```python id=clean-up
with psycopg.connect(autocommit=True) as conn:
    mine = conn.execute(
        "SELECT indexname FROM pg_indexes WHERE tablename = 'station_status' AND indexname LIKE 'ho\\_%'"
    ).fetchall()
    for (index,) in mine:
        conn.execute(f"DROP INDEX {index}")
tables = [t.table_id for t in BQ.list_tables(f"enjoytech-bq-lab.{SANDBOX}") if t.table_id.startswith("ho_")]
for table in tables:
    BQ.delete_table(f"enjoytech-bq-lab.{SANDBOX}.{table}")
print(f"dropped {len(mine)} index(es) and {len(tables)} table(s)")
```
