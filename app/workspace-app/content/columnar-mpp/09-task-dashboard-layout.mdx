```duckdb id=ref
CREATE OR REPLACE VIEW ref_t1 AS     -- T1: NYC, hour by hour, on 15 August
SELECT date_trunc('hour', captured_at) AS hour, avg(bikes_available) AS bikes
FROM 'station_status.parquet'
WHERE system_id = 'NYC' AND captured_at >= '2026-08-15' AND captured_at < '2026-08-16'
GROUP BY hour;

CREATE OR REPLACE VIEW ref_t2 AS     -- T2: station 6, hour by hour, over the whole quarter
SELECT date_trunc('hour', captured_at) AS hour, avg(bikes_available) AS bikes
FROM 'station_status.parquet'
WHERE station_id = 6
GROUP BY hour;

CREATE OR REPLACE VIEW ref_t3 AS     -- T3: the monthly average of each system
SELECT date_trunc('month', captured_at) AS month, system_id, avg(bikes_available) AS bikes
FROM 'station_status.parquet'
GROUP BY month, system_id;

SELECT (SELECT count(*) FROM ref_t1) AS t1_rows, (SELECT count(*) FROM ref_t2) AS t2_rows,
       (SELECT count(*) FROM ref_t3) AS t3_rows;
```

## Part A — your layout, on DuckDB (30 minutes)

Write the data again under names starting with `dash` (a file, or a folder of files), as many as the
tiles need. Every choice is yours: what to split by, what to sort by, what to add up in advance, and
whether one layout can serve all three tiles.

```duckdb id=layout manual
-- Your layout. Start from:
-- COPY (SELECT … FROM 'station_status.parquet' …)
--   TO 'dash' (FORMAT parquet, …);
```

Rewrite each tile against your layout, as a view named `tile_1`, `tile_2`, `tile_3`, with the same
columns as today's tile.

```duckdb id=tiles manual needs=layout
-- CREATE OR REPLACE VIEW tile_1 AS SELECT … ;
-- CREATE OR REPLACE VIEW tile_2 AS SELECT … ;
-- CREATE OR REPLACE VIEW tile_3 AS SELECT … ;
```

**Check 1: the same answers.** Each line must say `0`. The averages are compared to six decimals.

```duckdb needs=ref,tiles
WITH a1 AS (SELECT hour, round(bikes, 6) AS bikes FROM ref_t1), b1 AS (SELECT hour, round(bikes, 6) AS bikes FROM tile_1),
     a2 AS (SELECT hour, round(bikes, 6) AS bikes FROM ref_t2), b2 AS (SELECT hour, round(bikes, 6) AS bikes FROM tile_2),
     a3 AS (SELECT month, system_id, round(bikes, 6) AS bikes FROM ref_t3),
     b3 AS (SELECT month, system_id, round(bikes, 6) AS bikes FROM tile_3)
SELECT 'T1' AS tile, (SELECT count(*) FROM ((FROM a1 EXCEPT FROM b1) UNION ALL (FROM b1 EXCEPT FROM a1))) AS rows_that_differ
UNION ALL SELECT 'T2', (SELECT count(*) FROM ((FROM a2 EXCEPT FROM b2) UNION ALL (FROM b2 EXCEPT FROM a2)))
UNION ALL SELECT 'T3', (SELECT count(*) FROM ((FROM a3 EXCEPT FROM b3) UNION ALL (FROM b3 EXCEPT FROM a3)));
```

**Check 2: no tile reads more than today, and T1 and T3 read far less.** Compare `Data Read`, the
files scanned and the row groups scanned with today's tile; change `tile_1` to `tile_2` and `tile_3`.

```duckdb needs=tiles
EXPLAIN ANALYZE SELECT * FROM tile_1;
```

**Check 3: it stays small.** Your layout beside the one file.

```bash
du -sb station_status.parquet dash* 2>/dev/null    # bytes
```
