## A server, and every message to it

<Scene widget="agf-mcp-wire">
<Step>

The whole server. The cell writes it to a file, because a client starts it as a program of its own:

```python id=server
import pathlib

SERVER_SOURCE = '''\
import os

import psycopg
from mcp.server import MCPServer

AGENT_URL = os.environ["DATABASE_URL"].replace("postgres:postgres@", "agent_ro:agent_ro@")
mcp = MCPServer("pagila")


@mcp.tool()
def list_tables() -> dict:
    """List the tables of the Pagila database. Takes no arguments."""
    with psycopg.connect(AGENT_URL) as conn:
        found = conn.execute("""
            SELECT table_name FROM information_schema.tables
            WHERE table_schema = 'public' AND table_type = 'BASE TABLE'
              AND table_name NOT LIKE 'payment_p%' ORDER BY 1""").fetchall()
    return {"tables": [name for (name,) in found]}


@mcp.tool()
def describe_table(table_name: str) -> dict:
    """Columns and types of one table. Call it before writing SQL that reads the table."""
    with psycopg.connect(AGENT_URL) as conn:
        found = conn.execute("""
            SELECT column_name, data_type FROM information_schema.columns
            WHERE table_schema = 'public' AND table_name = %s
            ORDER BY ordinal_position""", (table_name,)).fetchall()
    if not found:
        return {"error": f"no such table: {table_name}", "hint": "call list_tables"}
    return {"table": table_name, "columns": [f"{name} {kind}" for name, kind in found]}


@mcp.tool()
def run_sql(query: str) -> dict:
    """Run one read-only PostgreSQL SELECT and return at most 50 rows."""
    try:
        with psycopg.connect(AGENT_URL) as conn:
            cur = conn.execute(query)
            found = cur.fetchmany(50)
            return {"columns": [c.name for c in cur.description],
                    "rows": [[str(v) for v in row] for row in found]}
    except psycopg.Error as e:
        return {"error": str(e).strip(), "hint": "fix the SQL; check names with describe_table"}
'''

RUN = '''

if __name__ == "__main__":
    mcp.run(transport="stdio")
'''
pathlib.Path("mcp_pagila_server.py").write_text(SERVER_SOURCE + RUN)
print(len(SERVER_SOURCE.splitlines()), "lines written to mcp_pagila_server.py")
```

`MCPServer` was called `FastMCP` before version 2 of the SDK; older examples use that name.

Be the client yourself: start the server, write one line of JSON to its stdin, read one line back.

```python
import json, subprocess, sys

server = subprocess.Popen([sys.executable, "mcp_pagila_server.py"], text=True,
                          stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)

def send(message, expect_reply=True):
    print("→", json.dumps(message))
    server.stdin.write(json.dumps(message) + "\n")
    server.stdin.flush()
    if expect_reply:
        print("←", server.stdout.readline())

send({"jsonrpc": "2.0", "id": 1, "method": "initialize",

send({"jsonrpc": "2.0", "method": "notifications/initialized"}, expect_reply=False)
```

## The agent finds its tools on the server

<Scene widget="agf-mcp-agent">
<Step>

`StdioServerParameters` says how to start the server, `ClientSession` speaks the protocol. The server
is a process of its own: the SDK starts it with only a few environment variables, so the database
address is passed on.

```python
import json, os, sys

from google import genai
from google.genai import types
from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import get_default_environment, stdio_client

print("a server's environment by default:", sorted(get_default_environment()))
SERVER = StdioServerParameters(command=sys.executable, args=["mcp_pagila_server.py"],
                               env={**get_default_environment(), "DATABASE_URL": os.environ["DATABASE_URL"]})


def declarations(listed):
    """An MCP tools/list result, as the function declarations Gemini takes."""
    return [types.FunctionDeclaration(name=tool.name, description=tool.description,
                                      parameters_json_schema=tool.input_schema) for tool in listed.tools]


async def catalog():
    async with stdio_client(SERVER, errlog=sys.__stderr__) as (read, write), ClientSession(read, write) as session:
        await session.initialize()
        return declarations(await session.list_tools())

for declaration in await catalog():
    print(declaration.name, "·", declaration.description, "·", declaration.parameters_json_schema.get("required", []))
```

</Step>
<Step>

The loop, with one change: a tool call goes to the server as `tools/call`. This cell calls the model.

```python
client = genai.Client(http_options={"timeout": 30_000, "retry_options": {"attempts": 2}})  # a call that hangs fails after 30 s, retried once
MODEL = os.environ["GEMINI_MODEL"]
SYSTEM = ("You answer questions about the Pagila DVD-rental database. Find tables and columns with the "
          "tools, then query with run_sql. State exact numbers from the result.")


async def ask(question, max_turns=8):
    async with stdio_client(SERVER, errlog=sys.__stderr__) as (read, write), ClientSession(read, write) as session:
        await session.initialize()
        config = types.GenerateContentConfig(
            system_instruction=SYSTEM,
            tools=[types.Tool(function_declarations=declarations(await session.list_tools()))],
            automatic_function_calling=types.AutomaticFunctionCallingConfig(disable=True))
        history = [types.Content(role="user", parts=[types.Part(text=question)])]
        for turn in range(max_turns):
            reply = await client.aio.models.generate_content(model=MODEL, contents=history, config=config)
            history.append(reply.candidates[0].content)
            if not reply.function_calls:
                return reply.text
            print(f"[model call {turn + 1}]")
            parts = []
            for call in reply.function_calls:
                result = await session.call_tool(call.name, dict(call.args))
                text = "\n".join(block.text for block in result.content)
                print(f"tools/call {call.name}({json.dumps(dict(call.args))})\n   → {' '.join(text.split())[:140]}")
                parts.append(types.Part.from_function_response(name=call.name, response={"result": text}))
            history.append(types.Content(role="user", parts=parts))
        raise RuntimeError(f"no answer after {max_turns} model calls")

print(await ask("Which film category has the most films, and how many?"))
```

The same question, by hand:
