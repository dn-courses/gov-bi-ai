The loop in three pieces: call the model, run the calls of one turn, go round.

```python
TOOL_IMPL = {"add": lambda a, b: a + b, "multiply": lambda a, b: a * b}
CALC = types.GenerateContentConfig(
    system_instruction="You are a careful calculator. Use the tools; do not do arithmetic in your head.",
    tools=[types.Tool(function_declarations=[ADD, MULTIPLY])],
    automatic_function_calling=OFF,
)


def call_model(contents):
    """One call to the model. Returns its message and the calls in it."""
    response = client.models.generate_content(model=MODEL, contents=contents, config=CALC)
    print(f"  {response.usage_metadata.prompt_token_count} tokens in")
    return response.candidates[0].content, response.function_calls or []


def run_calls(calls):
    """Run every call of one turn. Returns one user message: a response per call, in order."""
    parts = []
    for call in calls:
        result = TOOL_IMPL[call.name](**call.args)
        print(f"  {call.name}({call.args}) = {result}")
        parts.append(types.Part(function_response=types.FunctionResponse(id=call.id, name=call.name, response={"result": result})))
    return types.Content(role="user", parts=parts)


def run_agent(question, max_turns=10):
    contents = [types.Content(role="user", parts=[types.Part(text=question)])]
    for turn in range(max_turns):
        print(f"[turn {turn}]")
        reply, calls = call_model(contents)
        contents.append(reply)
        if not calls:
            return "".join(part.text or "" for part in reply.parts), contents
        contents.append(run_calls(calls))
    raise RuntimeError(f"agent did not finish in {max_turns} turns")
```
