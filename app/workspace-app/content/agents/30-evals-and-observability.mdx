```python id=reference needs=30a-task-2-the-warehouse-agent#layer
import yaml

REFERENCE_AGENT = r'''
"""The warehouse agent: answers questions about bike-share stations from the semantic model. It
never writes SQL, and a request that would read too much is refused before it runs."""

import os

from google.adk.agents import Agent
from pydantic import BaseModel

from . import semantic

# Traces go to Phoenix where there is one (the lab); a deployed agent has none beside it.
if os.environ.get("PHOENIX_COLLECTOR_ENDPOINT"):
    from phoenix.otel import register

    register(project_name="warehouse_agent", auto_instrument=True, batch=False, verbose=False)

MAX_BYTES = int(os.environ.get("MAX_BYTES", 1_000_000_000))


class Filter(BaseModel):
    """Keep only rows where a dimension has one of these values."""

    dimension: str
    values: list[str]


def list_metrics() -> dict:
    """List what can be asked: every metric and dimension with its meaning, the days the data covers,
    and the most bytes one query may read."""
    describe = lambda items: {name: item["description"] for name, item in items.items()}  # noqa: E731
    return {"metrics": describe(semantic.METRICS), "dimensions": describe(semantic.DIMENSIONS),
            "days": [str(semantic.FIRST), str(semantic.LAST)], "max_bytes": MAX_BYTES}


def query_metrics(metrics: list[str], dimensions: list[str], first_day: str, last_day: str,
                  filters: list[Filter] | None = None) -> dict:
    """Compute metrics over a range of days, optionally split by dimensions and filtered. The only
    way to get numbers. The query is priced first and refused if it would read too much.

    Args:
        metrics: metric names from list_metrics, at least one.
        dimensions: dimension names to split by; empty for one total.
        first_day: the first day, as YYYY-MM-DD.
        last_day: the last day, included, as YYYY-MM-DD.
        filters: dimension filters, e.g. one system.
    """
    try:
        sql, params = semantic.compile_query(metrics, dimensions, first_day, last_day,
                                             {f.dimension: f.values for f in filters or []})
    except ValueError as problem:
        return {"status": "error", "error": str(problem)}
    cost = semantic.price(sql, params)
    if cost > MAX_BYTES:
        return {"status": "refused", "bytes": cost, "max_bytes": MAX_BYTES,
                "hint": "Too much to read. Ask for fewer days, fewer dimensions or one system."}
    columns, rows = semantic.run(sql, params, MAX_BYTES)
    return {"status": "ok", "bytes": cost, "sql": sql, "columns": columns, "rows": rows}


root_agent = Agent(
    name="warehouse_agent",
    model=os.environ.get("GEMINI_MODEL", "gemini-3.5-flash-lite"),
    generate_content_config={"http_options": {"timeout": 30_000, "retry_options": {"attempts": 2}}},  # a call that hangs fails after 30 s, retried once
    instruction=(
        "You answer questions about bike-share stations. Numbers come only from query_metrics. "
        "Call list_metrics first to learn the names and the days the data covers. "
        "Every question needs a range of days; if the question gives none, ask for it. "
        "If a query is refused, narrow it and try once more, or say what to narrow. "
        "End every answer with the bytes the query read, in MB or GB. Never estimate a number."
    ),
    tools=[list_metrics, query_metrics],
)
'''.lstrip()

model = yaml.safe_load((FOLDER / "model.yml").read_text())
if not model["metrics"].get("empty_share", {}).get("sql"):
    print("model.yml has no empty_share yet: adding the reference one")
    model["metrics"]["empty_share"] = {"description": "Share of station readings with no bike to rent, from 0 to 1.",
                                       "sql": "ROUND(SAFE_DIVIDE(COUNTIF(num_bikes_available = 0), COUNT(num_bikes_available)), 4)"}
    (FOLDER / "model.yml").write_text(yaml.safe_dump(model, sort_keys=False, width=110))
mine = FOLDER / "agent.py"
text = mine.read_text() if mine.exists() else ""
if "root_agent" not in text or "NotImplementedError" in text or 'instruction=""' in text:
    print("agent.py is not finished yet: writing the reference one")
    mine.write_text(REFERENCE_AGENT)
semantic, agent = load("semantic"), load("agent")
print([tool.__name__ for tool in agent.root_agent.tools], list(semantic.METRICS))
```

```python id=setup
async def ask(root_agent, question):
    """Run one question; return the calls the agent made, the rows of its last good query, and its answer."""
    runner = InMemoryRunner(agent=root_agent, app_name="warehouse_agent")
    session = await runner.session_service.create_session(app_name="warehouse_agent", user_id="me")
    run = {"calls": [], "rows": None, "answer": ""}
    async for event in runner.run_async(user_id="me", session_id=session.id, run_config=RunConfig(max_llm_calls=8),
                                        new_message=types.Content(role="user", parts=[types.Part(text=question)])):
        run["calls"] += [(call.name, dict(call.args)) for call in event.get_function_calls()]
        for response in event.get_function_responses():
            if response.response.get("status") == "ok" and "rows" in response.response:
                run["rows"] = response.response["rows"]
        if event.is_final_response() and event.content and event.content.parts:
            run["answer"] = (event.content.parts[0].text or "").strip()
    return run
```

```python id=trajectory
from google.adk.evaluation.eval_case import IntermediateData, Invocation
from google.adk.evaluation.eval_metrics import EvalMetric
from google.adk.evaluation.trajectory_evaluator import TrajectoryEvaluator

trajectory = TrajectoryEvaluator(eval_metric=EvalMetric(
    metric_name="tool_trajectory_avg_score", threshold=1.0, criterion={"threshold": 1.0, "match_type": "IN_ORDER"}))


def invocation(calls):
    return Invocation(user_content=types.Content(role="user", parts=[types.Part(text="?")]),
                      intermediate_data=IntermediateData(tool_uses=[types.FunctionCall(name=n, args=a) for n, a in calls]))


def score(actual_calls, expected_calls):
    return trajectory.evaluate_invocations([invocation(actual_calls)], [invocation(expected_calls)]).overall_score


NYC = [{"dimension": "system", "values": ["lyft_nyc"]}]
EXPECTED = [("query_metrics", {"metrics": ["empty_share"], "dimensions": ["hour"],
                               "first_day": "2026-09-15", "last_day": "2026-09-15", "filters": NYC})]
for name, run in runs.items():
    print(f"{name:17} calls score:", score(run["calls"], EXPECTED))
```

## A trace is a table

Every run of the agent sends its spans to Phoenix: open http://localhost:6006 and pick the project
the lab's traces go to. The same spans come back over its REST API as rows.

```python id=trace
import time, urllib.request
from datetime import datetime

PHOENIX = "http://phoenix:6006/v1/projects"


def spans_of_last_run(app="warehouse_agent"):
    """The spans of the newest trace whose root is this agent's invocation, from any project."""
    spans = []
    for project in json.load(urllib.request.urlopen(PHOENIX))["data"]:
        spans += json.load(urllib.request.urlopen(f"{PHOENIX}/{project['id']}/spans?limit=1000"))["data"]
    roots = [s for s in spans if s["name"] == f"invocation [{app}]"]
    newest = max(roots, key=lambda s: s["start_time"])["context"]["trace_id"]
    return [s for s in spans if s["context"]["trace_id"] == newest]


await ask(agent.root_agent, QUESTION)
time.sleep(1)
spans = spans_of_last_run()
print(len(spans), "spans in trace", spans[0]["context"]["trace_id"])
```
