## The shape: names in, priced SQL out

The agent asks for metrics and dimensions over a range of days. The compiler writes the SQL, always
with the date filter on the partitioning column; a dry run prices it; above the limit it is refused.

"""The bike-share semantic layer: the model, a compiler from a request to the one statement that
answers it, the statement's price from a dry run, and a run that BigQuery itself caps."""

import datetime
from pathlib import Path

import yaml
from google.cloud import bigquery

MODEL = yaml.safe_load((Path(__file__).parent / "model.yml").read_text())
METRICS, DIMENSIONS = MODEL["metrics"], MODEL["dimensions"]
FIRST, LAST = (datetime.date.fromisoformat(str(day)) for day in MODEL["covers"])


def compile_query(metrics, dimensions, first_day, last_day, filters=None, limit=50):
    """The SQL for these metrics by these dimensions over these days, and the values it binds.
    Every statement carries the date range, on the column the table is partitioned by."""
    filters = filters or {}
    unknown = [n for n in metrics if n not in METRICS] + [n for n in [*dimensions, *filters] if n not in DIMENSIONS]
    if unknown or not metrics:
        raise ValueError(f"not in the model: {unknown}" if unknown else "ask for at least one metric")
    first, last = datetime.date.fromisoformat(first_day), datetime.date.fromisoformat(last_day)
    if not FIRST <= first <= last <= LAST:
        raise ValueError(f"days must lie between {FIRST} and {LAST}, first before last")
    select = [f"{DIMENSIONS[d]['sql']} AS {d}" for d in dimensions] + [f"{METRICS[m]['sql']} AS {m}" for m in metrics]
    where = [f"{MODEL['time']} >= TIMESTAMP(@first_day)",
             f"{MODEL['time']} < TIMESTAMP_ADD(TIMESTAMP(@last_day), INTERVAL 1 DAY)"]
    where += [f"CAST({DIMENSIONS[d]['sql']} AS STRING) IN UNNEST(@p{i})" for i, d in enumerate(filters)]
    groups = ", ".join(str(i + 1) for i in range(len(dimensions)))
    sql = "\n".join(
        [f"SELECT {', '.join(select)}", f"FROM `{MODEL['table']}`", f"WHERE {' AND '.join(where)}"]
        + ([f"GROUP BY {groups}", f"ORDER BY {len(dimensions) + 1} DESC"] if dimensions else [])
        + [f"LIMIT {int(limit)}"])
    params = [bigquery.ScalarQueryParameter("first_day", "DATE", first), bigquery.ScalarQueryParameter("last_day", "DATE", last)]
    params += [bigquery.ArrayQueryParameter(f"p{i}", "STRING", [str(v) for v in values]) for i, values in enumerate(filters.values())]
    return sql, params


def price(sql, params):
    """The bytes the statement would read: a dry run, which BigQuery does not bill."""
    config = bigquery.QueryJobConfig(query_parameters=params, dry_run=True, use_query_cache=False)
    return bigquery.Client().query(sql, job_config=config).total_bytes_processed


def run(sql, params, max_bytes):
    """Run the statement; BigQuery refuses it if it would bill more than max_bytes."""
    config = bigquery.QueryJobConfig(query_parameters=params, maximum_bytes_billed=max_bytes)
    result = bigquery.Client().query(sql, job_config=config).result()
    return [field.name for field in result.schema], [[str(v) for v in row.values()] for row in result]
'''.lstrip())
(FOLDER / "agent.py").exists() or (FOLDER / "agent.py").write_text("")


def load(name):
    """Import warehouse_agent.<name> as the files are now."""
    if str(FOLDER.parent) not in sys.path:
        sys.path.insert(0, str(FOLDER.parent))
    for module in [m for m in sys.modules if m.startswith("warehouse_agent")]:
        del sys.modules[module]
    return importlib.import_module(f"warehouse_agent.{name}")


semantic = load("semantic")
print("metrics:", list(semantic.METRICS), "| dimensions:", list(semantic.DIMENSIONS), "| days:", semantic.FIRST, "to", semantic.LAST)

**2. The gate and 3. the instruction.** `query_metrics` must price the query first, refuse it above
`MAX_BYTES` as data the model can read (with a hint what to narrow), and run it otherwise, returning
the bytes with the rows. Then write the instruction.

```python id=agent manual
(FOLDER / "agent.py").write_text(r'''
"""The warehouse agent: answers questions about bike-share stations from the semantic model. It
never writes SQL, and a request that would read too much is refused before it runs."""

import os

from google.adk.agents import Agent
from pydantic import BaseModel

from . import semantic

# Traces go to Phoenix where there is one (the lab); a deployed agent has none beside it.
if os.environ.get("PHOENIX_COLLECTOR_ENDPOINT"):
    from phoenix.otel import register

    register(project_name="warehouse_agent", auto_instrument=True, batch=False, verbose=False)

MAX_BYTES = int(os.environ.get("MAX_BYTES", 1_000_000_000))


class Filter(BaseModel):
    """Keep only rows where a dimension has one of these values."""

    dimension: str
    values: list[str]


def list_metrics() -> dict:
    """List what can be asked: every metric and dimension with its meaning, the days the data covers,
    and the most bytes one query may read."""
    describe = lambda items: {name: item["description"] for name, item in items.items()}  # noqa: E731
    return {"metrics": describe(semantic.METRICS), "dimensions": describe(semantic.DIMENSIONS),
            "days": [str(semantic.FIRST), str(semantic.LAST)], "max_bytes": MAX_BYTES}


def query_metrics(metrics: list[str], dimensions: list[str], first_day: str, last_day: str,
                  filters: list[Filter] | None = None) -> dict:
    """Compute metrics over a range of days, optionally split by dimensions and filtered. The only
    way to get numbers. The query is priced first and refused if it would read too much.

    Args:
        metrics: metric names from list_metrics, at least one.
        dimensions: dimension names to split by; empty for one total.
        first_day: the first day, as YYYY-MM-DD.
        last_day: the last day, included, as YYYY-MM-DD.
        filters: dimension filters, e.g. one system.
    """
    try:
        sql, params = semantic.compile_query(metrics, dimensions, first_day, last_day,
                                             {f.dimension: f.values for f in filters or []})
    except ValueError as problem:
        return {"status": "error", "error": str(problem)}
    # yours: price the statement with semantic.price(sql, params). Above MAX_BYTES, return
    # {"status": "refused", ...} with the bytes and a hint of what to narrow; nothing runs.
    # Otherwise run it with semantic.run(sql, params, MAX_BYTES) and return the status,
    # the bytes, the sql, the columns and the rows.
    raise NotImplementedError("the gate")


root_agent = Agent(
    name="warehouse_agent",
    model=os.environ.get("GEMINI_MODEL", "gemini-3.5-flash-lite"),
    generate_content_config={"http_options": {"timeout": 30_000, "retry_options": {"attempts": 2}}},  # a call that hangs fails after 30 s, retried once
    instruction="",  # yours: what it answers, where numbers come from, days, refusals, bytes
    tools=[list_metrics, query_metrics],
)
'''.lstrip())
agent = load("agent")
print([tool.__name__ for tool in agent.root_agent.tools])
```

## Ask it

From a cell, through ADK's runner. Each question calls the model a few times and reads at most the
limit.

```python id=ask
from google.adk.agents.run_config import RunConfig
from google.adk.runners import InMemoryRunner
from google.genai import types

runner = InMemoryRunner(agent=agent.root_agent, app_name="warehouse_agent")


async def ask(question):
    session = await runner.session_service.create_session(app_name="warehouse_agent", user_id="me")
    message = types.Content(role="user", parts=[types.Part(text=question)])
    print("Q:", question)
    async for event in runner.run_async(user_id="me", session_id=session.id, new_message=message,
                                        run_config=RunConfig(max_llm_calls=8)):
        for call in event.get_function_calls():
            print("   call:", call.name, dict(call.args))
        for response in event.get_function_responses():
            print("   ->", response.response.get("status"), response.response.get("bytes"), response.response.get("rows", [])[:2])
        if event.is_final_response() and event.content and event.content.parts:
            print("A:", (event.content.parts[0].text or "").strip(), "\n")


await ask("Which system had the highest share of empty stations on 2026-09-01?")
await ask("Compare the average bikes available per system over the first week of September 2026.")
```
