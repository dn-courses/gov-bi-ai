## When the SQL is wrong

The agent will write SQL that fails: a table that is not there, a column with another name. What
happens next depends on what the tool hands back. The prompt below is made worse on purpose: it
lists the tables under names they no longer have, as an out-of-date document would, and tells the
agent to go straight to SQL. `run_sql` now returns the whole message, and a hint from `hint_for`.

```python id=recover
def hint_for(error: psycopg.Error) -> str:
    return "fix the SQL and run it again"


def run_sql(query: str) -> dict:
    try:
        with psycopg.connect(AGENT_URL) as conn:
            cur = conn.execute(query)
            rows = cur.fetchmany(50)
            return {"columns": [c.name for c in cur.description],
                    "rows": [[None if v is None else str(v) for v in row] for row in rows],
                    "truncated": cur.fetchone() is not None}
    except psycopg.Error as e:
        return {"error": type(e).__name__, "message": str(e).strip(), "hint": hint_for(e)}


ALL_TOOLS = {"list_tables": list_tables, "describe_table": describe_table, "run_sql": run_sql}
STALE = ("You answer questions about Pagila, a PostgreSQL database. Its tables are "
         "movies(movie_id, title, length, rating), customer(customer_id, name, active) and "
         "payment(payment_id, customer_id, amount_total, paid_at). Use these names and go straight to run_sql. "
         "If a tool returns an error, read it and fix the query.")


def ask_stale(question: str, functions: dict | None = None, system: str = STALE, max_turns: int = 10) -> str:
    """The SQL agent with the tools in `functions` (all three when None). Prints every call and what came back."""
    functions = functions or ALL_TOOLS
    config = types.GenerateContentConfig(system_instruction=system,
                                         tools=[types.Tool(function_declarations=[d for d in declarations() if d.name in functions])],
                                         automatic_function_calling=types.AutomaticFunctionCallingConfig(disable=True))
    history = [types.Content(role="user", parts=[types.Part(text=question)])]
    for turn in range(max_turns):
        reply = client.models.generate_content(model=MODEL, contents=history, config=config)
        history.append(reply.candidates[0].content)
        if not reply.function_calls:
            print(f"[{turn + 1} model calls]")
            return reply.text
        results = []
        for call in reply.function_calls:
            try:
                result = functions[call.name](**(call.args or {}))
            except TypeError as e:   # arguments the declaration does not list
                result = {"error": "TypeError", "message": str(e), "hint": "call the tool with the arguments its declaration lists"}
            shown = str(call.args.get("query", call.args)).strip() if call.name == "run_sql" else f"{call.name}({', '.join(map(str, (call.args or {}).values()))})"
            print(shown, "\n  ->", json.dumps(result)[:160], "\n")
            results.append(types.Part.from_function_response(name=call.name, response=result))
        history.append(types.Content(role="user", parts=results))
    raise RuntimeError(f"no answer after {max_turns} model calls")
```

```python
def hint_for(error: psycopg.Error) -> str:
    if isinstance(error, psycopg.errors.InsufficientPrivilege):
        return "the database is read-only: do not retry; tell the user it cannot be done here"
    if isinstance(error, (psycopg.errors.UndefinedTable, psycopg.errors.UndefinedColumn)):
        return "a name is wrong: call list_tables or describe_table for the real names"
    return "fix the SQL and run it again"
```
