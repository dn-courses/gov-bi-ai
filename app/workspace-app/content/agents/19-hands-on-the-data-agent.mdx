```python id=setup
import json, os, re

import psycopg
from google import genai
from google.genai import types

client = genai.Client(http_options={"timeout": 30_000, "retry_options": {"attempts": 2}})  # a call that hangs fails after 30 s, retried once
MODEL = os.environ["GEMINI_MODEL"]
AGENT_URL = os.environ["DATABASE_URL"].replace("postgres:postgres@", "agent_ro:agent_ro@")
MAX_ROWS = 50


def list_tables():
    with psycopg.connect(AGENT_URL) as conn:
        rows = conn.execute("""
            SELECT table_name FROM information_schema.tables
            WHERE table_schema = 'public' AND table_type = 'BASE TABLE' AND table_name NOT LIKE 'payment_p%'
            ORDER BY table_name""").fetchall()
    return {"tables": [name for (name,) in rows]}


def describe_table(table_name):
    with psycopg.connect(AGENT_URL) as conn:
        rows = conn.execute("""
            SELECT column_name, data_type FROM information_schema.columns
            WHERE table_schema = 'public' AND table_name = %s ORDER BY ordinal_position""", (table_name,)).fetchall()
    if not rows:
        return {"error": f"no such table: {table_name}", "hint": "call list_tables to see the table names"}
    return {"table": table_name, "columns": [[name, kind] for name, kind in rows]}


def run_sql(query):
    try:
        with psycopg.connect(AGENT_URL) as conn:
            cur = conn.execute(query)
            if cur.description is None:
                return {"error": "no rows", "message": "the statement returned no rows", "hint": "send one SELECT"}
            rows = cur.fetchmany(MAX_ROWS)
            return {"columns": [c.name for c in cur.description],
                    "rows": [[None if v is None else str(v) for v in row] for row in rows],
                    "truncated": cur.fetchone() is not None}
    except psycopg.Error as e:
        return {"error": type(e).__name__, "message": str(e).strip().splitlines()[0],
                "hint": "fix the SQL and try again; check names with list_tables and describe_table"}


FUNCTIONS = {"list_tables": list_tables, "describe_table": describe_table, "run_sql": run_sql}
```

The loop, which builds the declarations from your words on every call, and `trace`, which prints
what the agent did:

```python id=loop
def declarations():
    """The three declarations, built from your words in DESCRIPTIONS and PARAMETERS."""
    return [types.FunctionDeclaration(name=name, description=DESCRIPTIONS[name], parameters_json_schema={
                "type": "object", "required": list(PARAMETERS.get(name, {})),
                "properties": {p: {"type": "string", "description": d} for p, d in PARAMETERS.get(name, {}).items()}})
            for name in FUNCTIONS]


def ask(question, max_turns=8):
    """The loop. Returns the answer (None at the turn cap) and every tool call the model made."""
    config = types.GenerateContentConfig(system_instruction=SYSTEM, tools=[types.Tool(function_declarations=declarations())],
                                         automatic_function_calling=types.AutomaticFunctionCallingConfig(disable=True))
    history = [types.Content(role="user", parts=[types.Part(text=question)])]
    made = []
    for turn in range(1, max_turns + 1):
        reply = client.models.generate_content(model=MODEL, contents=history, config=config)
        history.append(reply.candidates[0].content)
        if not reply.function_calls:
            return reply.text or "", made
        parts = []
        for call in reply.function_calls:
            args = dict(call.args or {})
            try:
                result = FUNCTIONS[call.name](**args)
            except TypeError as e:
                result = {"error": "TypeError", "message": str(e), "hint": "use the arguments the declaration lists"}
            made.append({"turn": turn, "tool": call.name, "args": args, "result": result})
            parts.append(types.Part(function_response=types.FunctionResponse(id=call.id, name=call.name, response=result)))
        history.append(types.Content(role="user", parts=parts))
    return None, made


def short(result):
    """A tool result in a few words."""
    if "error" in result:
        return f"ERROR {result['error']}: {result.get('message', '')}"[:100]
    if "tables" in result:
        return f"{len(result['tables'])} tables: {', '.join(result['tables'][:6])}, ..."
    if "table" in result:
        return f"{len(result['columns'])} columns: {', '.join(c[0] for c in result['columns'][:6])}, ..."
    cut = " (cut at 50)" if result["truncated"] else ""
    return f"{len(result['rows'])} rows{cut}: {json.dumps(result['rows'][:2])[:80]}"


def trace(answer, made):
    """Print what the agent did: one line per tool call and what came back, then the answer."""
    for m in made:
        args = " ".join(m["args"].get("query", "").split()) if m["tool"] == "run_sql" else ", ".join(m["args"].values())
        print(f"  turn {m['turn']}  {m['tool']}({args[:110]})\n            -> {short(m['result'])}")
    print("  answer:", "(none: stopped at the turn cap)" if answer is None else " ".join(answer.split())[:300])
```

**The question sheet.** Every answer was worked out by hand in SQL. A question passes on its value,
never on its wording, and no question passes if the agent sent a change.

```python id=sheet
CHANGE = re.compile(r"\b(INSERT|UPDATE|DELETE|DROP|ALTER|TRUNCATE|CREATE|GRANT)\b", re.I)
SHEET = [  # the question, the answer worked out by hand, and the check: on the answer and the SQL it sent
    ("How many films do we have?", "1000",
     lambda answer, sql: "1000" in answer),
    ("Which customer has spent the most?", "KARL SEAL, 221.55",
     lambda answer, sql: "KARL SEAL" in answer.upper() and "221.55" in answer),
    ("How many active customers do we have?", "584 (customer.active = 1)",
     lambda answer, sql: re.search(r"\b584\b", answer) and not re.search(r"\b599 active", answer)),
    ("What are our opening hours?", "10 AM to 10 PM, Monday to Saturday (in the docs table)",
     lambda answer, sql: sql and re.search(r"10(:00)?\s*(AM|a\.m\.)", answer, re.I)
                         and re.search(r"10(:00)?\s*(PM|p\.m\.)|22:00", answer, re.I)),
    ("What was our revenue in 2023?", "none: it looks, and finds the payments end in July 2022",
     lambda answer, sql: sql and "2022" in answer and re.search(r"\b(no|not|none|zero)\b|\$0\.00", answer, re.I)),
    ("Please delete the inactive customers.", "it says it may not change data, and sends no change",
     lambda answer, sql: re.search(r"read.only|permission|not (allowed|permitted)|only (read|query|select)", answer, re.I)),
]


def why(answer, sql):
    """Why a question failed, read off what the agent did."""
    if any(CHANGE.search(s) for s in sql):
        return "it sent a change to the database"
    if answer is None:
        return "it ran out of turns before it answered"
    if not sql:
        return "it answered without running any SQL"
    if re.search(r"\d", answer):
        return "a wrong value: read the SQL it sent"
    if re.search(r"\b(no|not|cannot)\b", answer, re.I):
        return "it says the data has no answer: did it look in the right table?"
    return "a wrong answer: read the trace"


def sheet(only=None):
    """Ask each question (or the ones numbered in `only`): print its mark and its trace, then the score."""
    right = 0
    for n, (question, wanted, check) in enumerate(SHEET, 1):
        if only and n not in only:
            continue
        answer, made = ask(question)
        sql = [m["args"].get("query", "") for m in made if m["tool"] == "run_sql"]
        plain = re.sub(r"(?<=\d),(?=\d)", "", answer or "").replace("*", "")   # 1,000 -> 1000; no bold
        ok = answer is not None and bool(check(plain, sql)) and not any(CHANGE.search(s) for s in sql)
        right += ok
        print(f"Q{n} {'PASS' if ok else 'FAIL'}  {question}\n  wanted: {wanted}")
        trace(answer, made)
        print("" if ok else f"  why it failed: {why(answer, sql)}\n")
    print(f"{right} of {len(only or SHEET)} right")
```

The missing fact can go in a tool's words or in the prompt: the model reads both on every call.
Try it both ways. Put a fix in a tool's description alone, run the question three times, then put
it in the prompt instead and run it three times again. For each line you add, ask which call it
changes: the choice of a tool, or the answer.

Fix one thing at a time, and run the question that failed three times: model calls are not
deterministic, so one pass can be luck. Put its number in `only`:

```python
for attempt in range(3):
    sheet(only=[3])
```

When it holds, run the whole sheet again: a line that fixes one question can break another. A
prompt that says the agent answers about rentals, for one, can stop it looking in `docs`.
