An agent can return a wrong answer that looks right. To find the mistake, you need to see what the
model was assuming when it picked its next step.

The SQL agent: three tools on Pagila, connected as the read-only role, and the loop. `agent()`

```python id=setup
import json, os
import psycopg
from google import genai
from google.genai import types

client = genai.Client()
MODEL = os.environ["GEMINI_MODEL"]
AGENT_URL = os.environ["DATABASE_URL"].replace("postgres:postgres@", "agent_ro:agent_ro@")


def list_tables():
    with psycopg.connect(AGENT_URL) as conn:
        rows = conn.execute("""SELECT table_name FROM information_schema.tables WHERE table_schema = 'public'
            AND table_type = 'BASE TABLE' AND table_name NOT LIKE 'payment_p%' ORDER BY table_name""").fetchall()
    return json.dumps([name for (name,) in rows])


def describe_table(table_name):
    with psycopg.connect(AGENT_URL) as conn:
        rows = conn.execute("""SELECT column_name, data_type FROM information_schema.columns
            WHERE table_schema = 'public' AND table_name = %s ORDER BY ordinal_position""", (table_name,)).fetchall()
    return json.dumps({"table": table_name, "columns": [{"name": n, "type": t} for n, t in rows]})


def run_sql(query):
    try:
        with psycopg.connect(AGENT_URL) as conn:
            cur = conn.execute(query)
            return json.dumps({"columns": [c.name for c in cur.description],
                               "rows": [[None if v is None else str(v) for v in r] for r in cur.fetchmany(50)]})
    except psycopg.Error as e:
        return json.dumps({"error": type(e).__name__, "message": str(e).strip()})


def declare(name, description, **params):
    return types.FunctionDeclaration(name=name, description=description, parameters_json_schema={
        "type": "object", "properties": params, "required": list(params)})


IMPL = {"list_tables": list_tables, "describe_table": describe_table, "run_sql": run_sql}
TOOLS = [declare("list_tables", "List the base tables of the Pagila database."),
         declare("describe_table", "The columns of one table, with their types.", table_name={"type": "string"}),
         declare("run_sql", "Run one read-only PostgreSQL SELECT; up to 50 rows come back as JSON.", query={"type": "string"})]
SYSTEM = "You answer questions about the Pagila film-rental database with the tools you are given."
QUESTION = "How many films has Susan Davis appeared in?"


def config(tools=TOOLS, **extra):
    return types.GenerateContentConfig(system_instruction=SYSTEM, tools=[types.Tool(function_declarations=tools)],
        automatic_function_calling=types.AutomaticFunctionCallingConfig(disable=True), **extra)


def agent(question, cfg=None):
    """The loop. Returns the message list and, per request, the usage the API reported."""
    history, usage = [types.Content(role="user", parts=[types.Part(text=question)])], []
    for _ in range(15):
        reply = client.models.generate_content(model=MODEL, contents=history, config=cfg or config())
        usage.append(reply.usage_metadata)
        history.append(reply.candidates[0].content)
        if not reply.function_calls:
            return history, usage
        history.append(types.Content(role="user", parts=[types.Part.from_function_response(name=c.name,
            response={"result": IMPL[c.name](**{k: v for k, v in c.args.items() if k != "reason"})})
            for c in reply.function_calls]))
    raise RuntimeError("no answer after 15 requests")


def answer(history):
    return "".join(p.text or "" for p in history[-1].parts if not p.thought).strip()


history, usage = agent(QUESTION)
print(answer(history))
```

def show(history):
    for i, message in enumerate(history):
        for p in message.parts:
            if p.thought:
                print(f"{i:>2} thought  {p.text.strip().splitlines()[0][:110]}")
            elif p.function_call:
                print(f"{i:>2} call     {p.function_call.name} {json.dumps(p.function_call.args)[:150]}")

## Two ways to make the reasoning visible

<Scene widget="agc-thought">
<Step>

Every tool gets a required `reason`. The loop drops it before calling the function, so the
functions do not change:

```python
REASON = {"type": "string", "description": "One sentence: why this call, and what you assume."}
WHY = [declare(d.name, d.description, reason=REASON, **d.parameters_json_schema["properties"]) for d in TOOLS]

reasoned, _ = agent(QUESTION, config(tools=WHY))
for message in reasoned:
    for p in message.parts:
        if p.function_call:
            print(f"{p.function_call.name:<15} reason: {p.function_call.args.get('reason')}")
print("\nanswer:", answer(reasoned))
```

On this model thinking is off unless you set a level. To see why an agent chose a step, switch it
on and read the summaries next to the calls. To keep a short note on every call in your own
records, require a `reason` argument. A note you asked for is not evidence that the model checked
anything.
