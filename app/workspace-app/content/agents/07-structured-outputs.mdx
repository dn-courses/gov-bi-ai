```python id=setup
import os, re

import psycopg
from google import genai
from google.genai import types

client = genai.Client(http_options={"timeout": 30_000, "retry_options": {"attempts": 2}})  # a call that hangs fails after 30 s, retried once
MODEL = os.environ["GEMINI_MODEL"]

with psycopg.connect(os.environ["DATABASE_URL"]) as conn:
    top = conn.execute("""SELECT c.customer_id, c.first_name || ' ' || c.last_name AS name, sum(p.amount) AS total
                          FROM payment p JOIN customer c USING (customer_id)
                          GROUP BY 1, 2 ORDER BY total DESC LIMIT 3""").fetchall()
FACTS = "\n".join(f"customer {cid}: {name}, {total} dollars" for cid, name, total in top)
QUESTION = f"Payments per customer, the top three:\n{FACTS}\n\nWho spent the most, and how much?"


def ask(**settings):
    return client.models.generate_content(model=MODEL, contents=QUESTION, config=types.GenerateContentConfig(**settings))


print(ask().text)
```

## How a call ends

<Scene widget="agb-ends">
<Step>

One function that names the ending, and three calls with the same schema: plain, with a tool the
model may ask for, and with the output cut at 8 tokens.

```python
LOOKUP = types.FunctionDeclaration(name="customer_rentals", description="Count the rentals of one customer, by customer id.",
    parameters_json_schema={"type": "object", "properties": {"customer_id": {"type": "integer"}}, "required": ["customer_id"]})


def ending(reply):
    """Which of the three ways this call ended."""
    if not reply.candidates:
        return f"stopped before it began: {reply.prompt_feedback.block_reason}"
    finish = reply.candidates[0].finish_reason
    if reply.function_calls:
        call = reply.function_calls[0]
        return f"a tool call: {call.name}({call.args})"
    if finish != types.FinishReason.STOP:
        return f"a stop: {finish.name}, text so far {reply.text!r}"
    return f"an answer: {reply.parsed!r}"


SCHEMA = dict(response_mime_type="application/json", response_schema=TopSpender)
TOOL = dict(tools=[types.Tool(function_declarations=[LOOKUP])],
            automatic_function_calling=types.AutomaticFunctionCallingConfig(disable=True))
print("schema           ->", ending(ask(**SCHEMA)))
print("schema + a tool  ->", ending(client.models.generate_content(
    model=MODEL, contents=QUESTION + "\nAlso: how many rentals has the top one made?",
    config=types.GenerateContentConfig(**SCHEMA, **TOOL))))
print("schema, 8 tokens ->", ending(ask(**SCHEMA, max_output_tokens=8)))
```
