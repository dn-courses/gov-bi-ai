```python id=setup
import psycopg

AGENT_URL = os.environ["DATABASE_URL"].replace("postgres:postgres@", "agent_ro:agent_ro@")


def as_agent(sql):
    """Run one statement as agent_ro and say what PostgreSQL answered."""
    try:
        with psycopg.connect(AGENT_URL) as conn:
            return conn.execute(sql).fetchone()
    except psycopg.Error as e:
        return f"{type(e).__name__}: {str(e).splitlines()[0]}"


for sql in ["SELECT count(*) FROM film",
            "INSERT INTO film (title, language_id) VALUES ('X', 1)",
            "DELETE FROM payment",
            "DROP TABLE film"]:
    print(f"{sql:<54} -> {as_agent(sql)}")
```

```python
def list_tables():
    """The base tables of the public schema, without payment's monthly partitions."""
    with psycopg.connect(AGENT_URL) as conn:
        rows = conn.execute("""
            SELECT table_name FROM information_schema.tables
            WHERE table_schema = 'public' AND table_type = 'BASE TABLE'
              AND table_name NOT LIKE 'payment_p%'
            ORDER BY table_name""").fetchall()
```

```python
def describe_table(table_name):
    """The columns of one table, in order: name, type, nullable."""
    with psycopg.connect(AGENT_URL) as conn:
        rows = conn.execute("""
            SELECT column_name, data_type, is_nullable FROM information_schema.columns
            WHERE table_schema = 'public' AND table_name = %s
            ORDER BY ordinal_position""", (table_name,)).fetchall()
    if not rows:
        return {"error": f"no such table: {table_name}", "hint": "call list_tables to see the table names"}
    return {"table": table_name, "columns": [[name, kind, nullable == "YES"] for name, kind, nullable in rows]}


print(describe_table("film"))
print(describe_table("movies"))
```

```python
MAX_ROWS = 50


def run_sql(query):
    """Run one statement as agent_ro; at most MAX_ROWS rows come back, every value as text."""
    try:
        with psycopg.connect(AGENT_URL) as conn:
            cur = conn.execute(query)
            if cur.description is None:
                return {"error": "the statement returned no rows", "hint": "send one SELECT"}
            rows = cur.fetchmany(MAX_ROWS)
            return {"columns": [c.name for c in cur.description],
                    "rows": [[None if v is None else str(v) for v in row] for row in rows],
                    "truncated": cur.fetchone() is not None}
    except psycopg.Error as e:
        return {"error": str(e).strip(), "hint": "fix the SQL and try again; check names with list_tables and describe_table"}


print(run_sql("SELECT count(*) AS rentals FROM rental"))
```

The declarations, and the table the loop looks each function up in:

```python
def string(description):
    return {"type": "string", "description": description}


def declare(name, description, required=(), **properties):
    return types.FunctionDeclaration(name=name, description=description, parameters_json_schema={
        "type": "object", "properties": properties, "required": list(required)})


TOOLS = [
    declare("list_tables", "List the tables of the Pagila film-rental database. Takes no arguments."),
    declare("describe_table", "Get the columns of one table, with their types. Call it before writing SQL that uses the table.",
            ["table_name"], table_name=string("Exact table name, as list_tables returns it.")),
    declare("run_sql", "Run one read-only PostgreSQL SELECT and return up to 50 rows. On failure returns error and hint.",
            ["query"], query=string("One PostgreSQL SELECT statement.")),
]
FUNCTIONS = {"list_tables": list_tables, "describe_table": describe_table, "run_sql": run_sql}


def configure():
    return types.GenerateContentConfig(tools=[types.Tool(function_declarations=TOOLS)],
                                       automatic_function_calling=types.AutomaticFunctionCallingConfig(disable=True))


CONFIG = configure()
```

```python
def ask(question, cap=8):
    """The loop over TOOLS. Prints each call and the start of its result; returns the answer."""
    history = [types.Content(role="user", parts=[types.Part(text=question)])]
    for _ in range(cap):
        reply = client.models.generate_content(model=MODEL, contents=history, config=CONFIG)
        history.append(reply.candidates[0].content)
        if not reply.function_calls:
            return reply.text
        parts = []
        for call in reply.function_calls:
            result = FUNCTIONS[call.name](**call.args)
            print(f"{call.name}({json.dumps(call.args)[:110]})\n   -> {json.dumps(result)[:110]}")
            parts.append(types.Part(function_response=types.FunctionResponse(id=call.id, name=call.name, response=result)))
        history.append(types.Content(role="user", parts=parts))
    return "(stopped at the turn cap)"


print("ANSWER:", ask("Show the revenue of each film category as a chart."))
```

```python
CHARTS = []


def make_chart(sql, kind, x, y, title):
    """Run the query, keep the chart for your code to draw; the model gets a short receipt."""
    result = run_sql(sql)
    if "error" in result:
        return result
    columns = result["columns"]
    if x not in columns or y not in columns:
        return {"error": f"x and y must be columns of the result: {columns}", "hint": "name them as the SELECT does"}
    try:
        points = [[row[columns.index(x)], float(row[columns.index(y)])] for row in result["rows"]]
    except (TypeError, ValueError):
        return {"error": f"{y} is not a number in every row", "hint": "y must be a numeric column"}
    CHARTS.append({"kind": kind, "title": title, "x": x, "y": y, "points": points})
    return {"chart": len(CHARTS), "drawn": f"a {kind} chart of {len(points)} points", "truncated": result["truncated"]}


TOOLS.append(declare("make_chart",
    "Draw the rows of one SELECT as a chart the person sees. Use it when the person asks for a chart, or when the answer is many numbers to compare.",
    ["sql", "kind", "x", "y", "title"],
    sql=string("One PostgreSQL SELECT returning a label column and a number column."),
    kind={"type": "string", "enum": ["bar", "line"], "description": "bar to compare categories, line for a trend over time."},
    x=string("The label column, as the SELECT names it."), y=string("The number column, as the SELECT names it."),
    title=string("A short title for the chart.")))
FUNCTIONS["make_chart"] = make_chart
CONFIG = configure()
```

```python
def show(chart, width=36):
    """The chart as text: one bar of blocks per point."""
    top = max(value for _, value in chart["points"])
    print(chart["title"])
    for label, value in chart["points"]:
        print(f"{label:>12}  {'█' * round(width * value / top):<{width}}  {value:,.2f}")
```
